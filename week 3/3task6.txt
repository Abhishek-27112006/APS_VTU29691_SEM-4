

class Solution {
    public String minRemoveToMakeValid(String s) {
        StringBuilder sb = new StringBuilder(s);
        java.util.Stack<Integer> stack = new java.util.Stack<>();
        
        // First pass: identify and mark invalid parentheses
        for (int i = 0; i < sb.length(); i++) {
            char c = sb.charAt(i);
            if (c == '(') {
                stack.push(i);
            } else if (c == ')') {
                if (!stack.isEmpty()) {
                    stack.pop();
                } else {
                    sb.setCharAt(i, '*'); 
                }
            }
        }
        
        // Any remaining indices in the stack are unmatched '('
        while (!stack.isEmpty()) {
            sb.setCharAt(stack.pop(), '*'); // Mark invalid '(' with a placeholder
        }
        
        // Build the final result by filtering out the placeholders
        StringBuilder result = new StringBuilder();
        for (int i = 0; i < sb.length(); i++) {
            char c = sb.charAt(i);
            if (c != '*') {
                result.append(c);
            }
        }
        
        return result.toString();
    }
}
